## プロジェクト概要
================================================================================

### 既存構造
- internal/           : 内部パッケージ（scanner, generator, parser, resolver 等）
- pkg/types/          : 公開型定義（Config, Port, Compose, Conflicts 等）
- cmd/                : CLI コマンド（root, up, clean, status）
- Makefile            : test-e2e ターゲットは ./test/e2e/... を想定

### 既存インターフェース
- scanner.PortDetector        : システムポート検出
- scanner.PortAllocator       : ポート割り当て
- scanner.UnifiedConflictDetector : 統一的衝突検知
- generator.OverrideGenerator : override.yml 生成
- parser.ComposeParser        : Docker Compose ファイル解析

### 設定構造（pkg/types/config.go）
- AppConfig: Port, File, Watcher, Log の設定を保持
- PortConfig: Range, Reserved, ExcludePrivileged
- PortRange: Start, End

### タスク 3.2: pkg/types/namespace.go 作成（Green フェーズ）

```go
// pkg/types/namespace.go
package types

import "fmt"

// NamespaceConfig は namespace の設定を表します。
// namespace は複数の開発者/環境間でポート範囲を分離するための機能です。
type NamespaceConfig struct {
    // Name は namespace の識別名です
    Name string `yaml:"name" json:"name"`

    // PortOffset は namespace 間のポート間隔です
    // 例: PortOffset=1000 の場合、namespace "dev" は 8000-8999、
    //     namespace "staging" は 9000-9999 を使用
    PortOffset int `yaml:"port_offset" json:"port_offset"`

    // PortRange は この namespace で使用するポート範囲です
    PortRange PortRange `yaml:"port_range" json:"port_range"`
}

// Validate は NamespaceConfig の妥当性を検証します。
func (c *NamespaceConfig) Validate() error {
    if c.Name == "" {
        return fmt.Errorf("namespace name is required")
    }

    if c.PortRange.Start > c.PortRange.End {
        return fmt.Errorf("invalid port range: start (%d) > end (%d)",
            c.PortRange.Start, c.PortRange.End)
    }

    if c.PortRange.Start < 1 || c.PortRange.End > 65535 {
        return fmt.Errorf("port range must be between 1 and 65535")
    }

    return nil
}

// Contains はポートが namespace の範囲内かどうかを確認します。
func (c *NamespaceConfig) Contains(port int) bool {
    return port >= c.PortRange.Start && port <= c.PortRange.End
}
```

```go
// test/e2e/namespace_isolation_test.go
package e2e

func TestNamespacePortIsolation(t *testing.T) {
    // テスト: 異なる namespace でポート範囲が分離されること
    //
    // シナリオ:
    // - namespace "dev" は 8000-8999 を使用
    // - namespace "staging" は 9000-9999 を使用
    // - 同じサービス名でも異なるポートが割り当てられること

    devNamespace := types.NamespaceConfig{
        Name:       "dev",
        PortOffset: 0,
        PortRange:  types.PortRange{Start: 8000, End: 8999},
    }

    stagingNamespace := types.NamespaceConfig{
        Name:       "staging",
        PortOffset: 1000,
        PortRange:  types.PortRange{Start: 9000, End: 9999},
    }

    // dev namespace でポート割り当て
    devPort := allocatePortForNamespace(t, devNamespace, "web")
    assert.True(t, devNamespace.Contains(devPort),
        "dev port should be in dev namespace range")

    // staging namespace でポート割り当て
    stagingPort := allocatePortForNamespace(t, stagingNamespace, "web")
    assert.True(t, stagingNamespace.Contains(stagingPort),
        "staging port should be in staging namespace range")

    // ポートが異なることを確認
    assert.NotEqual(t, devPort, stagingPort,
        "dev and staging should have different ports")
}
```

```go
func TestNamespaceConfigLoading(t *testing.T) {
    // テスト: .gopose.yaml から namespace 設定が読み込めること
    //
    // .gopose.yaml の構造例:
    // namespace:
    //   name: dev
    //   port_offset: 1000
    //   port_range:
    //     start: 8000
    //     end: 8999

    // 一時ディレクトリに .gopose.yaml を作成
    tmpDir := t.TempDir()
    configPath := filepath.Join(tmpDir, ".gopose.yaml")

    configContent := `
namespace:
  name: dev
  port_offset: 1000
  port_range:
    start: 8000
    end: 8999
`
    require.NoError(t, os.WriteFile(configPath, []byte(configContent), 0644))

    // 設定読み込み
    cfg, err := config.LoadFromFile(configPath)
    require.NoError(t, err)

    // namespace 設定の検証
    assert.Equal(t, "dev", cfg.Namespace.Name)
    assert.Equal(t, 1000, cfg.Namespace.PortOffset)
    assert.Equal(t, 8000, cfg.Namespace.PortRange.Start)
    assert.Equal(t, 8999, cfg.Namespace.PortRange.End)
}
```

### タスク 4.3: TestMultipleNamespaceConflictResolution（Red フェーズ）

```go
func TestMultipleNamespaceConflictResolution(t *testing.T) {
    // テスト: 複数 namespace 間での衝突解決
    //
    // シナリオ:
    // - 2つの namespace が同じホストで動作
    // - それぞれが同じポートを要求
    // - 衝突が検出され、各 namespace の範囲内で解決されること

    ctx := context.Background()

    // namespace 設定
    namespaces := []types.NamespaceConfig{
        {Name: "dev", PortRange: types.PortRange{Start: 8000, End: 8999}},
        {Name: "staging", PortRange: types.PortRange{Start: 9000, End: 9999}},
    }

    // 両方の compose ファイルで 8080 を要求
    composeConfig := createTestComposeConfig(t, map[string]int{
        "web": 8080,
        "api": 3000,
    })

    for _, ns := range namespaces {
        t.Run(ns.Name, func(t *testing.T) {
            // 衝突検出
            detector := createNamespaceAwareDetector(ns)
            conflicts, err := detector.DetectConflicts(ctx, composeConfig, ns.Name)
            require.NoError(t, err)

            // 衝突解決
            resolver := createNamespaceAwareResolver(ns)
            err = resolver.ResolveConflicts(ctx, conflicts)
            require.NoError(t, err)

            // 解決されたポートが namespace 範囲内であることを確認
            for _, conflict := range conflicts.PortConflicts {
                if conflict.Resolution != nil {
                    assert.True(t, ns.Contains(conflict.Resolution.ResolvedPort),
                        "resolved port %d should be in namespace %s range",
                        conflict.Resolution.ResolvedPort, ns.Name)
                }
            }
        })
    }
}
```
